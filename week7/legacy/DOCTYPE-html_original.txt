    <script>
        function sync(id) {
            document.getElementById('val_' + id).innerText = document.getElementById(id).value;
        }

        function updateStation() {
            document.getElementById('displayLocation').innerText = document.getElementById('stationSelect').value.split(',')[0];
        }

        function resetDefaults() {
            document.getElementById('pm10').value = 115;
            document.getElementById('temp').value = 24.5;
            document.getElementById('humidity').value = 68;
            document.getElementById('windSpeed').value = 2.1;
            document.getElementById('windDir').value = 80;
            document.getElementById('no2').value = 42.5;
            document.getElementById('so2').value = 14.2;

            ['pm10', 'temp', 'humidity', 'windSpeed', 'windDir'].forEach(sync);
            calculate();
        }

        function calculate() {
            const pm10 = parseFloat(document.getElementById('pm10').value) || 115;
            const humidity = parseFloat(document.getElementById('humidity').value) || 68;
            const temp = parseFloat(document.getElementById('temp').value) || 24.5;
            const no2 = parseFloat(document.getElementById('no2').value) || 42.5;

            // Live formula simulation matching Random Forest feature weighting
            let pm25 = (pm10 * 0.72) + (humidity * 0.28) + (no2 * 0.18) - (temp * 0.25);
            pm25 = Math.round(pm25 * 10) / 10;
            const aqi = Math.round(pm25 * 1.55);

            document.getElementById('predVal').innerHTML = `${pm25} <span class="text-base font-medium text-slate-400">µg/m³</span>`;
            document.getElementById('aqiVal').innerText = aqi;

            const box = document.getElementById('catBox');
            const title = document.getElementById('catTitle');
            const desc = document.getElementById('catDesc');
            const icon = document.getElementById('catIcon');

            if (pm25 <= 35.4) {
                box.className = "bg-emerald-950/30 p-6 rounded-2xl border border-emerald-800/50 flex flex-col justify-center space-y-2";
                title.className = "text-2xl font-bold text-emerald-400";
                title.innerText = "Good";
                desc.innerText = "Air quality is satisfactory, and air pollution poses little or no risk.";
                icon.className = "fa-solid fa-face-smile text-emerald-400 text-lg";
            } else if (pm25 <= 75.4) {
                box.className = "bg-amber-950/30 p-6 rounded-2xl border border-amber-800/50 flex flex-col justify-center space-y-2";
                title.className = "text-2xl font-bold text-amber-400";
                title.innerText = "Moderate";
                desc.innerText = "Air quality is acceptable; sensitive groups may experience minor health effects.";
                icon.className = "fa-solid fa-face-meh text-amber-400 text-lg";
            } else {
                box.className = "bg-rose-950/30 p-6 rounded-2xl border border-rose-800/50 flex flex-col justify-center space-y-2";
                title.className = "text-2xl font-bold text-rose-400";
                title.innerText = "Unhealthy";
                desc.innerText = "Everyone may begin to experience health effects; sensitive groups may experience serious effects.";
                icon.className = "fa-solid fa-head-side-mask text-rose-400 text-lg";
            }
        }
